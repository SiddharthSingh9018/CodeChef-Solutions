class Solution {public:    int totalFruits(vector<int>& fruits) {        unordered_map<int, int> count;        int L = 0;        int Maxf = 0;        int R = 0;                while (R < fruits.size()) {            count[fruits[R]]++;                                   while (count.size() > 2) {                count[fruits[L]]--;                if (count[fruits[L]] == 0) {                    count.erase(fruits[L]);                }                L++;            }                           Maxf = max(Maxf, R - L + 1);            R++;        }        return Maxf;    }};
