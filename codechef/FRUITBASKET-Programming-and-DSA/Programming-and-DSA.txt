using namespace std;class Solution {public:    int totalFruits(vector<int>& fruits) {        // write your code here         int sum=0;        int left=0;        int max =0;        unordered_map<int,int> freq;        for(int right=0;right<fruits.size();right++){            sum+=1;            freq[fruits[right]]++;            while(freq.size()>2){                sum-=1;                freq[fruits[left]]--;                if(freq[fruits[left]]==0){                    freq.erase(fruits[left]);                }                left++;            }            if(max<sum){                max =sum;            }        }        return max;    }};
