#include <bits/stdc++.h>using namespace std;int main() {    int N, M;    cin >> N >> M;    vector<tuple<int, int, int>> edges;    for (int i = 0; i < M; i++) {        int u, v, w;        cin >> u >> v >> w;        edges.push_back({u, v, w});    }    int S;    cin >> S;    const long long INF = MAX_INT;    vector<long long> dist(N, INF);    dist[S] = 0;    // Relax all edges N-1 times    for (int i = 0; i < N - 1; i++) {        bool changed = false;        for (auto [u, v, w] : edges) {            if (dist[u] != INF && dist[u] + w < dist[v]) {                dist[v] = dist[u] + w;                changed = true;            }        }        if (!changed) {            break;        }    }    for (int i = 0; i < N; i++) {        cout << dist[i];        if (i != N - 1) {            cout << " ";        }    }    return 0;}
