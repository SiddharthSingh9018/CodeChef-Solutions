#include <bits/stdc++.h>using namespace std;int main() {    ios::sync_with_stdio(false);    cin.tie(nullptr);    int tc;    cin >> tc;    while (tc--) {        int N, M;        cin >> N >> M;        string S, T;        cin >> S >> T;        vector<int> need(26, 0);        for (char c : T) {            need[c - 'a']++;        }        vector<int> have(26, 0);        int required = M;           int formed = 0;        int left = 0;        int ansLen = INT_MAX;        int ansStart = -1;        for (int right = 0; right < N; right++) {            char c = S[right];            have[c - 'a']++;                        if (have[c - 'a'] <= need[c - 'a']) {                formed++;            }                       while (formed == required) {                int len = right - left + 1;                if (len < ansLen) {                    ansLen = len;                    ansStart = left;                }                char remove = S[left];                have[remove - 'a']--;                if (have[remove - 'a'] < need[remove - 'a']) {                    formed--;                }                left++;            }        }        if (ansStart == -1) {            cout << -1 << endl;        }         else {            cout << S.substr(ansStart, ansLen) << endl;        }    }    return 0;}
