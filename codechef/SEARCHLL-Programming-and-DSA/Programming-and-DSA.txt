Node* removeKey(Node* head, int key) {    while(head != NULL && head->data == key) {        head = head->next;    }    if(head == NULL) return NULL;    Node* curr = head;    Node* Nextptr = head->next;    while(Nextptr != NULL) {        if(Nextptr->data == key) {            curr->next = Nextptr->next;            Nextptr = curr->next;             }        else {            curr = Nextptr;            Nextptr = Nextptr->next;        }    }    return head;}
