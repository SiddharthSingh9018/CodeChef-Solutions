#include <bits/stdc++.h>using namespace std;int main() {    int T;    cin >> T;    while (T--) {        int N, D;        cin >> N >> D;        vector<vector<pair<int,int>>> start(D + 2);        for (int i = 0; i < N; i++) {            int di, ti, si;            cin >> di >> ti >> si;            start[di].push_back({si, ti});        }              priority_queue<pair<int,int>> pq;         long long penalty = 0;        for (int day = 1; day <= D; day++) {                    for (auto &job : start[day]) {                pq.push(job);            }            if (!pq.empty()) {                auto [s, rem] = pq.top();                pq.pop();                rem--;                if (rem > 0) {                    pq.push({s, rem});                }            }        }        while (!pq.empty()) {            auto [s, rem] = pq.top();            pq.pop();            penalty += 1LL * s * rem;        }        cout << penalty << "\n";    }    return 0;}
