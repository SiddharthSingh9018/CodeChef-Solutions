class Solution {public:    long long calculateMaxRobbery(const vector<int> &houseValues) {        // write your code here         vector<long long> EMax(houseValues.size());        EMax[0] = houseValues[0];        EMax[1] = max(houseValues[1],houseValues[0]);                       for(int i=2;i<houseValues.size();i++){            EMax[i] = max(EMax[i-1],EMax[i-2]+houseValues[i]);        }                return EMax[houseValues.size()-1];    }};
