string isMinHeap(int A[], int N) {    for (int i = 0; i < N / 2; i++) {        // Safe check: if left exists and is smaller, OR if right exists and is smaller        if ((2 * i + 1 < N && A[i] > A[2 * i + 1]) || (2 * i + 2 < N && A[i] > A[2 * i + 2])) {            return "No";        }    }    return "Yes";}
